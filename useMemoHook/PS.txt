Create your own custom useMemo Hook.
useMemo is a react Hook that lets you cache the result of a calculation between re-renders.


const [counter, setCounter] = useState(0)
const [counter2, setCounter2] = useState(100)

const squaredValue = () => {
    console.log("expensive call")
    return counter * counter;
};

return (
    <div className='app'>
      <h2>Counter: {counter}</h2>
      <h2>Squared Counter: {squaredValue()}</h2>
      <button onClick={() => setCounter(counter + 1)}>Increment</button>

      <h2>Counter: {counter2}</h2>
      <button onClick={() => setCounter2(counter2 - 1)}>Decrement</button>
    </div>
  )

  ** this complete component is re-rendering when clicking decrement button 
  and squared counter (expensive calculation) takes place again. 
  It happens because if state changes in component it re-renders whole component. 


We can use useMemo hook
ex. cachedVal = useMemo(callBack(), dependencies) 

const App = () => {

  const [counter, setCounter] = useState(0)
  const [counter2, setCounter2] = useState(100)

  const squaredValue = () => {
    console.log("expensive call")
    return counter * counter;
  };

  const memoisedSquaredValue = useMemo(squaredValue, [counter])

  return (
    <div className='app'>
      <h2>Counter: {counter}</h2>
      <h2>Squared Counter: {memoisedSquaredValue}</h2>
      <button onClick={() => setCounter(counter + 1)}>Increment</button>

      <h2>Counter2: {counter2}</h2>
      <button onClick={() => setCounter2(counter2 - 1)}>Decrement</button>
    </div>
  )
}
